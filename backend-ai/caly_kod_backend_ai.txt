from fastapi import APIRouter
from app.services.finance_service import get_stock

router = APIRouter(prefix="/crypto", tags=["Crypto"])

CRYPTO_SYMBOLS = {
    "BTC": {"name": "Bitcoin", "ticker": "BTC-USD"},
    "ETH": {"name": "Ethereum", "ticker": "ETH-USD"},
    "USDT": {"name": "Tether", "ticker": "USDT-USD"},
    "BNB": {"name": "BNB", "ticker": "BNB-USD"},
    "SOL": {"name": "Solana", "ticker": "SOL-USD"},
}


@router.get(
    "/table",
    summary="Lista TOP kryptowalut do tabeli",
    description="Zwraca aktualne kursy i zmiany dla głównych kryptowalut pod tabelę na frontendzie."
)
def crypto_table():
    results = []
    for rank, (coin, info) in enumerate(CRYPTO_SYMBOLS.items(), start=1):
        try:
            data = get_stock(info["ticker"])
            d = data.model_dump() if hasattr(data, "model_dump") else dict(data)
            results.append({
                "rank": rank,
                "symbol": coin,
                "name": info["name"],
                "price": d.get("current_price", 0.0),
                "change24h": f"{d.get('change_percent', 0.0):+.2f}%",
                "marketCap": d.get("market_cap", "N/D")
            })
        except Exception:
            results.append({
                "rank": rank,
                "symbol": coin,
                "name": info["name"],
                "price": 0.0,
                "change24h": "0.00%",
                "marketCap": "N/D"
            })
    return results


@router.get(
    "/ai-impact",
    summary="Wpływ wydarzeń na rynek krypto (AI Impact)",
    description="Zwraca przeanalizowany wpływ ważnego wydarzenia na rynek kryptowalut."
)
def crypto_ai_impact():
    return {
        "event": "Nowe regulacje dotyczące rynku kryptowalut",
        "affectedMarkets": "BTC, ETH, USDT",
        "riskLevel": "Średnie",
        "sentiment": "Neutralny",
        "summary": "AI prognozuje zwiększoną zmienność na głównych kryptowalutach w obliczu zapowiedzi nowych ram prawnych.",
        "showChart": True
    }


@router.get(
    "/{symbol}",
    summary="Podstawowe dane pojedynczej kryptowaluty",
    description="Zwraca aktualną cenę i zmianę dla wybranej kryptowaluty."
)
def crypto_data(symbol: str):
    clean = symbol.upper().replace("-USD", "")
    info = CRYPTO_SYMBOLS.get(clean, {"name": clean, "ticker": f"{clean}-USD"})

    try:
        data = get_stock(info["ticker"])
        d = data.model_dump() if hasattr(data, "model_dump") else dict(data)
        return {
            "name": info["name"],
            "symbol": clean,
            "price": f"${d.get('current_price', 0.0):,.2f}",
            "change": f"{d.get('change_percent', 0.0):+.2f}%",
            "category": "crypto"
        }
    except Exception:
        return {
            "name": clean,
            "symbol": clean,
            "price": "Brak danych",
            "change": "0.00%",
            "category": "crypto"
        }

def get_secret() -> bytes:
    DATA_DIR.mkdir(parents=True, exist_ok=True)
    if not SECRET_PATH.exists():
        try:
            fd = os.open(SECRET_PATH, os.O_CREAT | os.O_EXCL | os.O_WRONLY, 0o600)
        except FileExistsError:
            pass
        else:
            with os.fdopen(fd, "wb") as f:
                f.write(secrets.token_bytes(32))
    return SECRET_PATH.read_bytes()


def ensure_users_file():
    DATA_DIR.mkdir(parents=True, exist_ok=True)
    if not USERS_PATH.exists():
        wb = Workbook()
        sheet = wb.active
        sheet.title = "Users"
        sheet.append(["user_id", "name", "email", "password_hash", "created_at"])
        wb.save(USERS_PATH)


def get_users():
    ensure_users_file()
    wb = load_workbook(USERS_PATH, read_only=True, data_only=True)
    try:
        sheet = wb["Users"]
        records = []
        for row in sheet.iter_rows(min_row=2, values_only=True):
            if row[0] and row[2]:
                records.append(dict(zip(
                    ["user_id", "name", "email", "password_hash", "created_at"], row[:5]
                )))
        return records
    finally:
        wb.close()


def append_user(user):

def hash_password(password: str) -> str:
    salt = secrets.token_bytes(16)
    rounds = 600_000
    derived = hashlib.pbkdf2_hmac("sha256", password.encode("utf-8"), salt, rounds)
    return f"pbkdf2_sha256${rounds}${salt.hex()}${derived.hex()}"


def check_password(password: str, stored: str) -> bool:
    try:
        scheme, count, salt_hex, digest_hex = str(stored).split("$")
        if scheme != "pbkdf2_sha256":
            return False
        check = hashlib.pbkdf2_hmac(
            "sha256", password.encode("utf-8"), bytes.fromhex(salt_hex), int(count)
        )
        return hmac.compare_digest(check, bytes.fromhex(digest_hex))
    except (ValueError, TypeError):
        return False


def b64url(data: bytes) -> str:
    return base64.urlsafe_b64encode(data).rstrip(b"=").decode("ascii")


def unb64url(value: str) -> bytes:
    return base64.urlsafe_b64decode(value + "=" * (-len(value) % 4))


def make_token(user_id: str) -> str:
    data = {"sub": user_id, "iat": int(time.time()), "exp": int(time.time()) + SESSION_SECONDS}
    payload = b64url(json.dumps(data, separators=(",", ":")).encode())
    signature = b64url(hmac.new(get_secret(), payload.encode(), hashlib.sha256).digest())
    return f"{payload}.{signature}"


def token_to_user_id(token: str) -> str:
    try:
        payload, signature = token.split(".")
        expected = b64url(hmac.new(get_secret(), payload.encode(), hashlib.sha256).digest())
        if not hmac.compare_digest(signature, expected):
            raise ValueError("invalid signature")
        data = json.loads(unb64url(payload))
        if data.get("exp", 0) <= int(time.time()) or not isinstance(data.get("sub"), str):
            raise ValueError("expired or invalid")
        return data["sub"]
    except (ValueError, TypeError, KeyError, json.JSONDecodeError):
        raise HTTPException(status_code=401, detail="Sesja wygasła. Zaloguj się ponownie.")

@router.post("/register", status_code=status.HTTP_201_CREATED)
def register(body: RegisterBody):
    email = body.email.strip().lower()
    if "@" not in email or "." not in email.rsplit("@", 1)[-1]:
        raise HTTPException(status_code=422, detail="Podaj poprawny adres e-mail.")
    with LOCK:
        if any(str(u["email"]).lower() == email for u in get_users()):
            raise HTTPException(status_code=409, detail="Konto z tym adresem już istnieje.")
        user = {
            "user_id": str(uuid.uuid4()),
            "name": body.name.strip(),
            "email": email,
            "password_hash": hash_password(body.password),
            "created_at": datetime.now(timezone.utc).isoformat(),
        }
        append_user(user)
    return {"user": public_user(user), "access_token": make_token(user["user_id"]), "token_type": "bearer"}


@router.post("/login")
def login(body: LoginBody):
    email = body.email.strip().lower()
    with LOCK:
        user = next((u for u in get_users() if str(u["email"]).lower() == email), None)
    if not user or not check_password(body.password, str(user["password_hash"])):
        raise HTTPException(status_code=401, detail="Nieprawidłowy e-mail lub hasło.")
    return {"user": public_user(user), "access_token": make_token(str(user["user_id"])), "token_type": "bearer"}


def current_user(authorization: str | None = Header(default=None)):
    if not authorization or not authorization.startswith("Bearer "):
        raise HTTPException(status_code=401, detail="Wymagane logowanie.")
    user_id = token_to_user_id(authorization[len("Bearer "):])
    with LOCK:
        user = next((u for u in get_users() if str(u["user_id"]) == user_id), None)
    if not user:
        raise HTTPException(status_code=401, detail="Nie znaleziono użytkownika.")
    return public_user(user)

def get_macro_indicator(series_key: str, limit: int = 12) -> dict:
    """
    Pobiera najświeższe odczyty z FRED API (sort_order=desc)
    oraz historię obserwacji dla warstw Data Lake.
    """
    cfg = SERIES_CONFIG.get(series_key.upper(), {"id": series_key, "units": "lin"})
    series_id = cfg["id"]
    units = cfg.get("units", "lin")

    url = "https://api.stlouisfed.org/fred/series/observations"
    params = {
        "series_id": series_id,
        "api_key": settings.FRED_API_KEY.strip() if settings.FRED_API_KEY else "",
        "file_type": "json",
        "sort_order": "desc",  # Pobieramy od najnowszych
        "units": units,
        "limit": limit
    }
    
    try:
        response = requests.get(url, params=params, timeout=5)
        response.raise_for_status()
        data = response.json().get("observations", [])
        
        valid_obs = [obs for obs in data if obs.get("value") and obs.get("value") != "."]
        
        if valid_obs:
            latest = valid_obs[0]  # Przy sort_order=desc indeks 0 to najświeższy odczyt
            val_float = round(float(latest["value"]), 2)
            
            history_list = [
                {"date": o["date"], "value": round(float(o["value"]), 2)}
                for o in reversed(valid_obs)
            ]
            
            return {
                "date": latest["date"],
                "value": str(val_float),
                "history": history_list
            }
    except Exception as e:
        print(f"[FRED API WARNING] Nie udało się pobrać {series_key} ({series_id}): {e}")
        
    return {"date": "N/A", "value": "N/A", "history": []}

def _send_reset_email(email: str, link: str) -> None:
    api_key = os.environ.get(

        ),

        logger.warning(

    logger.error(

    raise RuntimeError(
        f"Resend rejected email: HTTP {response.status_code}"
    )

@router.post("/forgot-password")
def forgot_password(body: ForgotPasswordBody):
    email = body.email.strip().lower()

    with LOCK:
        now = time.time()

        if (
            now - RESET_HISTORY.get(email, 0)
            < RESET_COOLDOWN_SECONDS
        ):
            return {
                "message": PUBLIC_MESSAGE
            }

        RESET_HISTORY[email] = now

        user = next(
            (
                u
                for u in get_users()
                if str(u["email"]).strip().lower() == email
            ),
            None,
        )

        if user is None:
            return {
                "message": PUBLIC_MESSAGE
            }

        if (
            not os.environ.get("RESEND_API_KEY")
            or not os.environ.get("RESEND_FROM_EMAIL")
        ):
            logger.error(
                "Wysyłanie wiadomości resetującej "
                "nie zostało skonfigurowane."
            )

            return {
                "message": PUBLIC_MESSAGE
            }

        token = secrets.token_urlsafe(32)

        token_hash = hashlib.sha256(
            token.encode("utf-8")
        ).hexdigest()

        tokens = _read_tokens()

        tokens = {
            key: value
            for key, value in tokens.items()
            if (
                float(value.get("expires", 0)) > now
                and str(value.get("user_id"))
                != str(user["user_id"])
            )
        }

        tokens[token_hash] = {
            "user_id": str(user["user_id"]),
            "expires": now + RESET_TTL_SECONDS,
        }

        _atomic_write_tokens(tokens)

    # ========================================================
    # GENEROWANIE LINKU DO FORMULARZA
    # ========================================================

    frontend_url = os.environ.get(
        "FRONTEND_URL",
        "http://localhost:5173",
    ).rstrip("/")

        f"?token={quote(token)}"
    )

    # ========================================================
    # WYSYŁANIE WIADOMOŚCI
    # ========================================================

    try:
        _send_reset_email(email, link)

    except Exception:
        logger.exception(
            "Nie udało się wysłać wiadomości resetującej."
        )

        with LOCK:
            tokens = _read_tokens()
            tokens.pop(token_hash, None)
            _atomic_write_tokens(tokens)

    return {
        "message": PUBLIC_MESSAGE
    }

@router.post("/reset-password")
def reset_password(body: ResetPasswordBody):
    token_hash = hashlib.sha256(
        body.token.encode("utf-8")
    ).hexdigest()

    with LOCK:
        tokens = _read_tokens()

        entry = tokens.get(token_hash)

        if (
            not entry
            or float(entry.get("expires", 0)) <= time.time()
        ):
            raise HTTPException(
                status_code=400,
                detail=(
                    "Link jest nieprawidłowy lub wygasł. "
                    "Poproś o nowy."
                ),
            )

        user_id = str(entry["user_id"])

        users = get_users()

        if not any(
            str(user["user_id"]) == user_id
            for user in users
        ):
            raise HTTPException(
                status_code=400,
                detail=(
                    "Link jest nieprawidłowy lub wygasł. "
                    "Poproś o nowy."
                ),
            )

        new_hash = hash_password(body.password)

        # ====================================================
        # ZAPIS NOWEGO HASŁA W EXCELU
        # ====================================================

        wb = load_workbook(USERS_PATH)

        try:
            sheet = wb["Users"]

            changed = False

            for row in sheet.iter_rows(min_row=2):
                if str(row[0].value) == user_id:
                    row[3].value = new_hash
                    changed = True
                    break

            if not changed:
                raise HTTPException(
                    status_code=400,
                    detail="Nie znaleziono konta.",
                )

            with tempfile.NamedTemporaryFile(
                dir=DATA_DIR,
                suffix=".xlsx",
                delete=False,
            ) as file:
                tmp = Path(file.name)

            try:
                wb.save(tmp)
                os.replace(tmp, USERS_PATH)

            finally:
                tmp.unlink(missing_ok=True)

        finally:
            wb.close()

        # Unieważnienie wykorzystanych tokenów resetowania.
        tokens = {
            key: value
            for key, value in tokens.items()
            if str(value.get("user_id")) != user_id
        }

        _atomic_write_tokens(tokens)

    return {
        "message": (
            "Hasło zostało zmienione. "
            "Możesz się zalogować."
        )
    }

def _detect_symbols(message: str, fallback_symbol: str | None) -> list[str]:
    symbols = []
    if fallback_symbol and fallback_symbol.lower() not in ["string", "none", ""]:
        symbols.append(fallback_symbol.upper())

    words = re.findall(r"\b[a-zA-Z0-9.-]+\b", message.lower())
    for word in words:
        if word in TICKER_MAP and TICKER_MAP[word] not in symbols:
            symbols.append(TICKER_MAP[word])

    return symbols[:2]


def process_message(request: ChatRequest) -> ChatResponse:
    target_symbols = _detect_symbols(request.message, request.symbol)

    market_context = ""
    for sym in target_symbols:
        try:
            stock_data = get_stock(sym)
            data_dict = stock_data.model_dump() if hasattr(stock_data, "model_dump") else dict(stock_data)
            market_context += f"- Dane dla {sym}: {data_dict}\n"
        except Exception:
            pass

    custom_system_instruction = (
        f"{SYSTEM_PROMPT}\n\n"
        "ŚCISŁE ZASADY ODPOWIEDZI:\n"
        "1. ZWIĘZŁOŚĆ: Odpowiedź ma mieć maksymalnie 3-4 zdania lub zwartą listę punktową.\n"

    )

    if market_context:
        custom_system_instruction += f"\nAKTUALNE DANE Z SYSTEMU DLA SPÓŁEK:\n{market_context}\n"

    contents = []
    if request.history:
        for msg in request.history:
            contents.append(
                types.Content(
                    role=msg.role,
                    parts=[types.Part.from_text(text=msg.content)]
                )
            )

    contents.append(
        types.Content(
            role="user",
            parts=[types.Part.from_text(text=request.message)]
        )
    )

    # Bezpieczne wywołanie z obsługą błędu 503 / przeciążenia Gemini
    try:
        response = client.models.generate_content(
            model="gemini-3.5-flash-lite",

                system_instruction=custom_system_instruction,
                max_output_tokens=300,
                temperature=0.2,

def _json_serializer(obj):
    """Pomocniczy konwerter typów nieobsługiwanych standardowo przez bibliotekę json."""
    if isinstance(obj, (datetime, date, pd.Timestamp)):
        return obj.isoformat()
    return str(obj)

                existing_articles = json.load(f)
        except Exception:
            existing_articles = []

    # Deduplikacja po unikalnym URL
    seen_urls = {a.get("url") for a in articles if a.get("url")}
    combined = list(articles)
    for old_art in existing_articles:
        if old_art.get("url") not in seen_urls:
            combined.append(old_art)
            seen_urls.add(old_art.get("url"))

    # Sortowanie od najnowszych i zachowanie max 50 artykułów
    combined.sort(key=lambda x: x.get("published_at", ""), reverse=True)
    combined = combined[:50]

    with open(file_path, "w", encoding="utf-8") as f:
        json.dump(combined, f, ensure_ascii=False, indent=2, default=_json_serializer)

def _hash_id(url: str) -> str:
    return hashlib.md5(url.strip().encode("utf-8")).hexdigest()


def _translate_titles_batch(titles: list[str]) -> list[str]:
    """Tłumaczy nagłówki na język polski w jednym wsadowym zapytaniu do Gemini."""
    if not titles:
        return []

    prompt = (
        "Przetłumacz poniższe nagłówki wiadomości finansowych i geopolitycznych na naturalny, "
        "rzeczowy język polski. Zwróć wynik WYŁĄCZNIE jako poprawną tablicę JSON, "
        "dokładnie w tej samej kolejności: [\"tytuł 1\", \"tytuł 2\", ...].\n\n"
        f"Nagłówki:\n{json.dumps(titles, ensure_ascii=False)}"
    )

    try:
        response = client.models.generate_content(
            model="gemini-3.5-flash-lite",
            contents=prompt,
            config=genai.types.GenerateContentConfig(
                response_mime_type="application/json",
                temperature=0.1,
            ),
        )
        translated = json.loads(response.text)
        if isinstance(translated, list) and len(translated) == len(titles):
            return translated
    except Exception as e:
        logger.warning(f"Błąd tłumaczenia tytułów przez Gemini: {e}")

    return titles


def fetch_from_rss(category_key: str) -> list[dict]:
    """Pobiera wiadomości z polskich kanałów RSS."""
    feeds = RSS_SOURCES.get(category_key, [])
    now_iso = datetime.now(timezone.utc).isoformat()
    articles = []

    for feed_url, source_name in feeds:
        try:
            parsed = feedparser.parse(feed_url)
            for entry in parsed.entries[:25]:
                url = getattr(entry, "link", "").strip()
                title = getattr(entry, "title", "").strip()
                if not url or not title:
                    continue

                pub = getattr(entry, "published", now_iso)
                desc = getattr(entry, "summary", "")

                articles.append({
                    "id": _hash_id(url),
                    "title": title,
                    "description": desc,
                    "source": source_name,
                    "url": url,
                    "image_url": None,
                    "category": category_key,
                    "published_at": pub,
                    "fetched_at": now_iso,
                    "language": "pl",
                })
        except Exception as e:
            logger.error(f"Błąd pobierania RSS {feed_url}: {e}")

    return articles


def fetch_and_process_category(category_key: str) -> list[dict]:
    """
    Pobiera artykuły. Polskie RSS są priorytetem (0 tokenów),
    a NewsAPI stanowi uzupełnienie tylko, jeśli brakuje artykułów.
    """
    now_iso = datetime.now(timezone.utc).isoformat()
    clean_cat = category_key.strip().lower().replace(" ", "_")
    all_articles = []

    # 1. Priorytet: Zaufane polskie źródła RSS
    rss_articles = fetch_from_rss(clean_cat)
    all_articles.extend(rss_articles)

    # 2. Uzupełnienie z NewsAPI tylko wtedy, gdy z RSS jest mniej niż 15 artykułów
    if len(all_articles) < 15:
        query = CATEGORY_MAPPING.get(clean_cat, clean_cat)
        raw_articles = []
        try:
            raw_articles = get_news(query)
            if not raw_articles and clean_cat in CATEGORY_MAPPING:
                raw_articles = get_news("finance")
        except Exception as e:
            logger.warning(f"NewsAPI niedostępne dla {clean_cat}: {e}")

        if raw_articles:
            try:
                save_to_bronze("newsapi", clean_cat, {"articles": raw_articles})
            except Exception:
                pass

        filtered_newsapi = []
        for art in raw_articles:
            source_name = (art.get("source", {}).get("name") or "").lower()
            url = (art.get("url") or "").lower()
            title = art.get("title") or ""

            if not title or "[Removed]" in title:
                continue
            if any(domain in url or domain in source_name for domain in EXCLUDED_DOMAINS):
                continue

            filtered_newsapi.append(art)

        selected_newsapi = filtered_newsapi[:5]
        orig_titles = [a.get("title", "") for a in selected_newsapi]
        translated = _translate_titles_batch(orig_titles)

        for art, trans_title in zip(selected_newsapi, translated):
            url = art.get("url", "")
            all_articles.append({
                "id": _hash_id(url),
                "title": trans_title,
                "description": art.get("description", ""),
                "source": art.get("source", {}).get("name", "NewsAPI"),
                "url": url,
                "image_url": art.get("urlToImage"),
                "category": clean_cat,
                "published_at": art.get("publishedAt", now_iso),
                "fetched_at": now_iso,
                "language": "pl",
            })

    # 3. Zapis do warstwy Silver Data Lake (deduplikacja i sortowanie)
    if all_articles:
        save_news_to_silver(clean_cat, all_articles)

    return all_articles
